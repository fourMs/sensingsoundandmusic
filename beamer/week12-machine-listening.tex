\begin{frame}{Aims}
  \begin{itemize}
    \item Pipeline: audio $\rightarrow$ preprocessing $\rightarrow$ features $\rightarrow$ tasks (classification, segmentation, similarity).
    \item MIR landscape — datasets, metrics, limits (genre bias, Western centrism).
    \item Compare machine outputs to human listening claims responsibly.
  \end{itemize}
\end{frame}

\begin{frame}{Roadmap}
  \begin{enumerate}
    \item Features: spectral, tonal, temporal (what librosa-like tools expose).
    \item Higher-level: chords, structure, source separation (ideas).
    \item Neural networks: layers, weights, training; why the features stop being ours to choose.
    \item ML overview — supervised vs unsupervised; evaluation pitfalls.
    \item Ethics: consent, surveillance, environmental cost, transparency (AI use in coursework).
  \end{enumerate}
\end{frame}

\begin{frame}{Human vs machine}
  \begin{itemize}
    \item Machines optimise proxies — ``ground truth'' labels encode cultural assumptions.
    \item Useful for exploration + assisted analysis; not a theory of musical meaning alone.
  \end{itemize}
\end{frame}

\begin{frame}{Course integration}
  \begin{itemize}
    \item Psychoacoustics \& harmony weeks explain what features \emph{approximate}.
    \item Electroacoustics explains sampling artefacts affecting algorithms.
  \end{itemize}
\end{frame}

\begin{frame}{In-class}
  \begin{itemize}
    \item Inspect one feature plot (chromagram/onset strength) for a familiar track.
    \item Discussion: where could bias hurt musical fairness?
  \end{itemize}
\end{frame}

\begin{frame}{Closing}
