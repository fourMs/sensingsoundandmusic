\begin{frame}{Aims}
  \begin{itemize}
    \item Ear pathway from outer ear to cortex (big picture).
    \item Loudness, pitch, timbre — subjective correlates of physical dimensions.
    \item Masking, streaming, illusions — prediction vs violation.
  \end{itemize}
\end{frame}

\begin{frame}{Roadmap}
  \begin{enumerate}
    \item Anatomy refresher; thresholds and dynamic range.
    \item Critical bands; simultaneous vs forward masking (ideas).
    \item Pitch theories (place vs timing — high level).
    \item Auditory scenes: streaming and continuity (` gestalt').
    \item Demo discussion (masking / continuity / speech analogies).
  \end{enumerate}
\end{frame}

\begin{frame}{Core message}
  \begin{itemize}
    \item Perception is \textbf{not} a linear meter of physics — context and attention reshape ``what is heard''.
    \item JNDs and uncertainty inform musical mixing and orchestration.
  \end{itemize}
\end{frame}

\begin{frame}{Link forward}
  \begin{itemize}
    \item Electroacoustics: how codecs and monitors preserve or discard cues.
    \item Machine listening: features approximate perceptual correlates (later weeks).
  \end{itemize}
\end{frame}

\begin{frame}{In-class}
  \begin{itemize}
    \item Predict outcomes for simple masking/streaming demos (hypothesis $\rightarrow$ listen).
    \item Relate one illusion to a concert or headphone listening situation.
  \end{itemize}
\end{frame}

\begin{frame}{Outside class}
  \begin{itemize}
    \item Chapter exercises; revise terminology for thresholds and scales (phon/sone overview if covered).
  \end{itemize}
\end{frame}

\begin{frame}{Summary}
  \begin{itemize}
    \item Psychoacoustics bridges signal and experience — essential for any tech-mediated music.
    \item Next: \textbf{Electroacoustics} — microphones, ADC/DAC, monitors, signal chain.
  \end{itemize}
\end{frame}
