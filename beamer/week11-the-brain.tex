\begin{frame}{Aims}
  \begin{itemize}
    \item Auditory pathway landmarks — cochlea to cortex (schematic).
    \item Music cognition: expectation, memory, emotion — high-level framing.
    \item Methods trade-offs: EEG, MEG, fMRI, fNIRS (what each resolves well/poorly).
  \end{itemize}
\end{frame}

\begin{frame}{Roadmap}
  \begin{enumerate}
    \item Tonotopic organisation; hierarchical processing (idea).
    \item Distributed networks for pitch, timing, timbre, syntax (not one ``music area'').
    \item Emotion \& aesthetics — reward, memory, appraisal (conceptual).
    \item BCIs / examples — responsible hype-check.
  \end{enumerate}
\end{frame}

\begin{frame}{Reading empirical work}
  \begin{itemize}
    \item Claim $\rightarrow$ evidence $\rightarrow$ method $\rightarrow$ limits (course checklist).
    \item Group averages vs individuals; lab stimuli vs ecological validity.
  \end{itemize}
\end{frame}
