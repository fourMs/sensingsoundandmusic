\begin{frame}{Aims}
  \begin{itemize}
    \item Compare \textbf{hearing} and \textbf{listening}; practise vocabulary for sound.
    \item Survey fields: sound studies, acoustics, psychoacoustics (preview).
    \item Introduce soundscape thinking and influential practitioners.
  \end{itemize}
\end{frame}

\begin{frame}{Roadmap}
  \begin{enumerate}
    \item Describing sounds — dimensions and metaphors.
    \item Soundscapes \& acoustic ecology (Schafer; soundwalking).
    \item Sound objects / artistic experiments (brief).
    \item Activities: listening exercises + reflection.
  \end{enumerate}
\end{frame}

\begin{frame}{Three analytical lenses (preview)}
  \begin{itemize}
    \item \textbf{Physical}: signals, spectra, levels (links to Acoustics week).
    \item \textbf{Perceptual}: pitch, loudness, streams (Psychoacoustics week).
    \item \textbf{Cultural / ecological}: meaning in context (soundscape studies).
  \end{itemize}
\end{frame}

\begin{frame}{Soundscape keywords}
  \begin{itemize}
    \item Keynote, signal, soundmark; hi-fi vs lo-fi environments.
    \item Soundwalking / deep listening as deliberate practice.
    \item Ethical stance: who gets heard in accounts of a place?
  \end{itemize}
\end{frame}

\begin{frame}{Development \& enculturation (hook)}
  \begin{itemize}
    \item Listening expertise is learned and culturally shaped.
    \item Expectations change what counts as ``musical'' structure (revisited in harmony/rhythm).
  \end{itemize}
\end{frame}

\begin{frame}{In-class}
  \begin{itemize}
    \item Two-minute closed-eyes listen in place; note layers and labels.
    \item Share one keyword for the ``feel'' of the soundscape.
  \end{itemize}
\end{frame}

\begin{frame}{Outside class}
  \begin{itemize}
    \item Reading + sound journal entry (short).
    \item Optional: record 60 seconds of environment; annotate salient events.
  \end{itemize}
\end{frame}

\begin{frame}{Summary}
  \begin{itemize}
    \item Multiple valid languages for sound — choose purpose-fit descriptions.
    \item Next: \textbf{Acoustics} — physical causes of vibration, propagation, rooms, instruments.
  \end{itemize}
\end{frame}
